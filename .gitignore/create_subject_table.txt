CREATE TABLE SUBJECT (
    subject_code SMALLINT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(50) NOT NULL,

    marks_theory TINYINT UNSIGNED NOT NULL,
    marks_practical TINYINT UNSIGNED DEFAULT NULL,
    marks_internal_assessment TINYINT UNSIGNED DEFAULT NULL,

    CONSTRAINT chk_valid_marks CHECK (
        -- exactly one of PA or IA is non-NULL
        (
            (marks_practical IS NOT NULL AND marks_internal_assessment IS NULL)
            OR
            (marks_practical IS NULL AND marks_internal_assessment IS NOT NULL)
        )
        AND
        -- valid ranges
        marks_theory BETWEEN 0 AND 100
        AND
        (marks_practical IS NULL OR marks_practical BETWEEN 0 AND 100)
        AND
        (marks_internal_assessment IS NULL OR marks_internal_assessment BETWEEN 0 AND 100)
        AND
        -- total = 100
        (
            (marks_practical IS NOT NULL AND marks_theory + marks_practical = 100)
            OR
            (marks_internal_assessment IS NOT NULL AND marks_theory + marks_internal_assessment = 100)
        )
    )
);


CREATE TABLE SUBJECT(

subject_code SMALLINT UNSIGNED,
course_name VARCHAR(50) NOT NULL,

marks_theory TINYINT UNSIGNED NOT NULL,
marks_practical TINYINT UNSIGNED DEFAULT NULL,
marks_internal_assessment TINYINT UNSIGNED DEFAULT NULL,

PRIMARY KEY(subject_code),

CONSTRAINT chk_valid_marks CHECK(
    -- exactly one of PA or IA is non-NULL
    (
        -- marks_practical
        (
            (
            marks_practical IS NOT NULL
            AND
            marks_practical BETWEEN 0 AND 100
            )
            AND marks_internal_assessment IS NULL
        )
        OR
        -- marks_internal_assessment
        (
            (
            marks_internal_assessment IS NOT NULL
            AND
            marks_internal_assessment BETWEEN 0 AND 100
            )
            AND marks_practical IS NULL)
        )
    )
    AND
    -- marks_theory
    marks_theory BETWEEN 0 AND 100
    AND
    (
        (marks_practical IS NOT NULL AND marks_theory + marks_practical = 100)
        OR
        (marks_internal_assessment IS NOT NULL AND marks_theory + marks_internal_assessment = 100)
    )
)
);
