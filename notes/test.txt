import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.title("CBSE Layout Example")
root.geometry("800x250")

# ==============================
# MAIN OUTER FRAME (big box)
# ==============================
main_frame = ttk.Frame(root, padding=10, relief="solid")
main_frame.grid(row=0, column=0, sticky="nsew")

# Allow resizing behavior
root.columnconfigure(0, weight=1)
root.rowconfigure(0, weight=1)

# ==============================
# SUB-FRAMES INSIDE MAIN FRAME
# ==============================

photo1_frame = ttk.Frame(main_frame, width=100, height=100, relief="solid")
text_frame   = ttk.Frame(main_frame, padding=10, relief="solid")
photo2_frame = ttk.Frame(main_frame, width=100, height=100, relief="solid")
button_frame = ttk.Frame(main_frame, padding=5, relief="solid")

# Grid placement (left → right)
photo1_frame.grid(row=0, column=0, padx=5)
text_frame.grid(row=0, column=1, padx=5, sticky="nsew")
photo2_frame.grid(row=0, column=2, padx=5)
button_frame.grid(row=0, column=3, padx=5)

# Let text frame expand horizontally
main_frame.columnconfigure(1, weight=1)

# ==============================
# CONTENT INSIDE EACH FRAME
# ==============================

# Photo 1 placeholder
photo1_label = ttk.Label(photo1_frame, text="Photo 1\n100x100")
photo1_label.place(relx=0.5, rely=0.5, anchor="center")

# Center text label
text_label = ttk.Label(
    text_frame,
    text="Central Board of Secondary Education",
    wraplength=300,
    anchor="center",
    justify="center"
)
text_label.pack(expand=True, fill="both")

# Photo 2 placeholder
photo2_label = ttk.Label(photo2_frame, text="Photo 2\n100x100")
photo2_label.place(relx=0.5, rely=0.5, anchor="center")

# Buttons stacked vertically
btn1 = ttk.Button(button_frame, text="Button 1")
btn2 = ttk.Button(button_frame, text="Button 2")

btn1.pack(fill="x", pady=5)
btn2.pack(fill="x", pady=5)

# ==============================
# OPTIONAL INFO TEXT BELOW
# ==============================

info_label = ttk.Label(
    root,
    text="Photo 1: 100x100 | Photo 2: 100x100",
    padding=5
)
info_label.grid(row=1, column=0, sticky="w")

root.mainloop()
